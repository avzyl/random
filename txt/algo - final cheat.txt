import json
import datetime

# Load users and notes from JSON files
with open("users.json", "r") as f:
  users = json.load(f)
with open("notes.json", "r") as f:
  notes = json.load(f)

def add_user(username, password, first_name, last_name):
  # Check if user already exists
  for user in users:
    if user["username"] == username:
      print("Error: User already exists.")
      return

  # Add new user
  users.append({
    "username": username,
    "password": password,
    "first_name": first_name,
    "last_name": last_name,
    "notes_ids": []
  })
  print("User added successfully.")

def delete_user(username):
  # Find user and delete
  for i, user in enumerate(users):
    if user["username"] == username:
      del users[i]
      print("User deleted successfully.")
      return
  print("Error: User not found.")

def update_user(username, first_name=None, last_name=None, password=None):
  # Find user and update details
  for user in users:
    if user["username"] == username:
      if first_name is not None:
        user["first_name"] = first_name
      if last_name is not None:
        user["last_name"] = last_name
      if password is not None:
        user["password"] = password
      print("User updated successfully.")
      return
  print("Error: User not found.")

def add_note(username, title, message):
  # Find user and add note
  for user in users:
    if user["username"] == username:
      # Assign unique ID to new note
      note_id = 1
      if len(notes) > 0:
        note_id = max([note["id"] for note in notes]) + 1
      # Add note and update user's notes IDs
      notes.append({
        "id": note_id,
        "title": title,
        "message": message,
        "date_time": datetime.datetime.now().strftime("%m/%d/%Y %I:%M %p")
      })
      user["notes_ids"].append(note_id)
      print("Note added successfully.")
      return
  print("Error: User not found.")

def update_note(id, title=None, message=None):
  # Find note and update details
  for note in notes:
    if note["id"] == id:
      if title is not None:
        note["title"] = title
      if message is not None:
        note["message"] = message
      print("Note updated successfully.")
      return
  print("Error: Note not found.")
